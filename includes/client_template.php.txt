<?php
/**
 * Cloaking SaaS - Client (upload to YOUR landing server)
 *
 * Configured values are embedded below. Keep this file server-side only.
 */

define('CLOAK_CLIENT_CREDENTIAL', '{{CLIENT_CREDENTIAL}}');
define('CLOAK_CAMPAIGN_ID', '{{CAMPAIGN_ID}}');
define('CLOAK_VERIFY_URL', '{{VERIFY_URL}}');

function cloak_is_https(): bool
{
    return !empty($_SERVER['HTTPS']) && $_SERVER['HTTPS'] !== 'off';
}

function cloak_fail(string $message, int $status = 502): never
{
    http_response_code($status);
    exit('<!DOCTYPE html><html><body>' . htmlspecialchars($message, ENT_QUOTES, 'UTF-8') . '</body></html>');
}

function cloak_verify(array $fingerprint, string $visitorToken, string $visitorCookie): array
{
    $params = $_GET;
    unset($params['_fph'], $params['_fv'], $params['_debug']);

    $body = json_encode([
        'campaign_id' => CLOAK_CAMPAIGN_ID,
        'ip' => $_SERVER['REMOTE_ADDR'] ?? '',
        'user_agent' => $_SERVER['HTTP_USER_AGENT'] ?? '',
        'referer' => $_SERVER['HTTP_REFERER'] ?? '',
        'language' => $_SERVER['HTTP_ACCEPT_LANGUAGE'] ?? '',
        'accept' => $_SERVER['HTTP_ACCEPT'] ?? '',
        'host' => $_SERVER['HTTP_HOST'] ?? '',
        'params' => $params,
        'fingerprint' => $fingerprint,
        'visitor_token' => $visitorToken,
        'visitor_cookie' => $visitorCookie,
    ]);

    $ctx = stream_context_create(['http' => [
        'method' => 'POST',
        'header' => "Content-Type: application/json\r\nAuthorization: Bearer " . CLOAK_CLIENT_CREDENTIAL . "\r\n",
        'content' => $body,
        'timeout' => 5,
        'ignore_errors' => true,
    ]]);

    $res = @file_get_contents(CLOAK_VERIFY_URL, false, $ctx);
    if ($res === false) {
        cloak_fail('Service unavailable', 503);
    }

    $data = json_decode($res, true);
    if (!is_array($data) || !array_key_exists('allowed', $data)) {
        cloak_fail('Verification failed', 502);
    }

    return $data;
}

function cloak_render_iframe(string $target): never
{
    $safe = htmlspecialchars($target, ENT_QUOTES, 'UTF-8');
    echo "<!DOCTYPE html>\n<html>\n<head>\n<meta charset=\"UTF-8\">\n"
       . "<meta name=\"viewport\" content=\"width=device-width, initial-scale=1, maximum-scale=1, user-scalable=0\">\n"
       . "<style>html,body,iframe{margin:0;padding:0;height:100%;width:100%;overflow:hidden}</style>\n"
       . "</head>\n<body>\n"
       . "<iframe src=\"{$safe}\" style=\"position:fixed;top:0;left:0;bottom:0;right:0;width:100%;height:100%;border:none\" "
       . "allowfullscreen=\"allowfullscreen\" webkitallowfullscreen=\"webkitallowfullscreen\" mozallowfullscreen=\"mozallowfullscreen\"></iframe>\n"
       . "</body>\n</html>";
    exit;
}

function cloak_render_redirect(string $target, string $redirectType, int $delay): never
{
    if ($redirectType === 'meta' || $delay > 0) {
        $safe = htmlspecialchars($target, ENT_QUOTES, 'UTF-8');
        echo "<!DOCTYPE html>\n<html>\n<head>\n"
           . "<meta charset=\"UTF-8\">\n"
           . "<meta http-equiv=\"refresh\" content=\"{$delay};url={$safe}\">\n"
           . "<script>setTimeout(function(){window.location.href="
           . json_encode($target) . ";}, {$delay}000);</script>\n"
           . "</head>\n<body style=\"font-family:sans-serif;text-align:center;padding:4rem\">\n"
           . "<p>Redirecting…</p>\n</body>\n</html>";
        exit;
    }

    $status = $redirectType === '301' ? 301 : ($redirectType === '303' ? 303 : 302);
    header('Location: ' . $target, true, $status);
    exit;
}

function cloak_render_reject_error(int $code): never
{
    $status = in_array($code, [400, 403, 404, 410, 429, 451], true) ? $code : 403;
    http_response_code($status);
    echo '<!DOCTYPE html><html><body style="font-family:sans-serif;text-align:center;padding:4rem">'
       . '<h1>' . $status . '</h1><p>Access denied.</p></body></html>';
    exit;
}

function cloak_is_safe_local_target(string $target): bool
{
    if ($target === '' || $target !== basename($target) || preg_match('/[\/\\\\]/', $target)) {
        return false;
    }

    return !str_starts_with($target, '.');
}

// ---- Read fingerprint payload from the tracker round-trip ----------------------
$fph = (string) ($_GET['_fph'] ?? '');
$fv = (string) ($_GET['_fv'] ?? '');
$fingerprint = [];
if ($fph !== '') {
    $raw = base64_decode(strtr($fph, '-_', '+/'), true);
    $decoded = is_string($raw) ? json_decode($raw, true) : null;
    $fingerprint = is_array($decoded) ? $decoded : [];
}

$resp = cloak_verify($fingerprint, $fv, (string) ($_COOKIE['clk'] ?? ''));

// ---- The server needs fingerprint data: run one JS round-trip ------------------
if (!empty($resp['fingerprint_required'])) {
    $path = parse_url($_SERVER['REQUEST_URI'] ?? '/', PHP_URL_PATH) ?: '/';
    $query = $_GET;
    unset($query['_fph'], $query['_fv'], $query['_debug']);
    $qs = http_build_query($query);
    $redirect = $path . ($qs !== '' ? '?' . $qs : '');
    $cfg = json_encode(['redirect' => $redirect], JSON_HEX_TAG | JSON_HEX_AMP | JSON_HEX_APOS | JSON_HEX_QUOT);
    echo "<!DOCTYPE html>\n<html>\n<head>\n<meta charset=\"UTF-8\">\n"
       . "<meta name=\"viewport\" content=\"width=device-width, initial-scale=1.0\">\n"
       . "<title>Loading…</title>\n"
       . "<script>window.__CLOAK_CFG__ = {$cfg};</script>\n"
       . "<script src=\"tracker.min.js\"></script>\n"
       . "</head>\n<body style=\"font-family:sans-serif;text-align:center;padding:4rem\">\n"
       . "<p>Please wait…</p>\n</body>\n</html>";
    exit;
}

// ---- Allowed: issue the visitor token, then redirect or show the money page ----
if (!empty($resp['allowed'])) {
    $visitorCookie = (string) ($resp['visitor_cookie'] ?? '');
    if ($fv !== '' && $visitorCookie !== '') {
        setcookie('clk', $visitorCookie, [
            'expires' => time() + 86400 * 365,
            'path' => '/',
            'secure' => cloak_is_https(),
            'httponly' => true,
            'samesite' => 'Lax',
        ]);
    }

    $target = (string) ($resp['pass_target'] ?? '');
    if (!empty($resp['forward_utms']) && preg_match('#^https?://#i', $target)) {
        $params = $_GET;
        unset($params['_fph'], $params['_fv'], $params['_debug']);
        if ($params !== []) {
            $target .= (strpos($target, '?') !== false ? '&' : '?') . http_build_query($params);
        }
    }

    if (preg_match('#^https?://#i', $target)) {
        if (($resp['offer_method'] ?? 'redirect') === 'iframe') {
            cloak_render_iframe($target);
        }

        cloak_render_redirect(
            $target,
            (string) ($resp['redirect_type'] ?? '302'),
            (int) ($resp['redirect_delay'] ?? 0)
        );
    }

    if (!cloak_is_safe_local_target($target)) {
        http_response_code(404);
        exit('<!DOCTYPE html><html><body>Page file missing: ' . htmlspecialchars($target, ENT_QUOTES, 'UTF-8') . '</body></html>');
    }

    $file = __DIR__ . '/' . $target;
    if (is_file($file)) {
        readfile($file);
        exit;
    }

    http_response_code(404);
    exit('<!DOCTYPE html><html><body>Page file missing: ' . htmlspecialchars($target, ENT_QUOTES, 'UTF-8') . '</body></html>');
}

// ---- Denied ----------------------------------------------------------------
$rejectMode = (string) ($resp['reject_mode'] ?? 'white');
if ($rejectMode === 'error') {
    cloak_render_reject_error((int) ($resp['reject_code'] ?? 403));
}
$rejectTarget = (string) ($resp['reject_target'] ?? '');
if (preg_match('#^https?://#i', $rejectTarget)) {
    header('Location: ' . $rejectTarget, true, 302);
    exit;
}
if ($rejectTarget !== '') {
    echo $rejectTarget;
    exit;
}
http_response_code(404);
exit;
